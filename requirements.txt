# Python dependencies for Corruption Isolation Engine (CIE)
#
# Install:  python3 -m pip install -r requirements.txt
#           ./install_dependencies.sh          (checks Python/FFmpeg too)
#
# Everything here is a normal, installable PyPI package. The previous file
# listed `tkinter`, `sqlite3`, `zipfile36` and `PyPDF4`, which cannot be
# installed with pip (tkinter/sqlite3 ship with CPython, zipfile36 targets
# Python < 3.6, and PyPDF4 was abandoned before PyPDF2 was forked into pypdf),
# so `pip install -r requirements.txt` failed outright and took the whole
# documented setup path down with it.
#
# Core engine -------------------------------------------------------------
# (the CLI works with *no* third-party packages at all; these upgrade the
#  depth of validation and the speed of the maths)

Pillow>=9.0.0            # image decoding: JPEG/PNG/BMP/TIFF/WEBP validation
pypdf>=4.0.0             # PDF parsing (successor of PyPDF2 >= 3.0)
python-docx>=0.8.11      # DOCX package validation
openpyxl>=3.0.9          # XLSX workbook validation

# Recommended -------------------------------------------------------------
numpy>=1.21.0            # floating-point histogram maths in cie_math (pure
                         # Python fallback exists, so this stays optional)

# Optional ----------------------------------------------------------------
python-pptx>=0.6.21      # PPTX validation (without it, PPTX falls back to the
                         # generic OOXML/zip checks)
# FFmpeg is a *binary*, not a pip package: install it with your OS package
# manager to enable ffprobe-based media validation (mkv/mp4/mp3/...).
#   Debian/Ubuntu: sudo apt-get install ffmpeg
#   macOS:         brew install ffmpeg
#   Windows:       winget install Gyan.FFmpeg
# ffmpeg-python is deliberately NOT required: the validator calls ffprobe
# directly.

# Development / tests -----------------------------------------------------
pytest>=7.0.0            # the suite in tests/
